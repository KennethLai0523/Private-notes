9. local variable inside a method within a class 
class Scope{
    String classVariable; // class-level

    void classMethod(){
        String localVariable; // local-level
    }

    void run(){
        localVariable = "Doesn't work! cannot find symbol"; // it causes an error
        // because localVariable doesn't exist within this scope

        classVariable = "assign text to class field/attribute";
        // this one works because the variable exists one step outside of the local brackets (i.e. class scope)
    }
}

class Main{
    public static void main(String[] args){
        Scope scope = new Scope();
        scope.run();
    }
}
